if(NOT WIN32)
  message(STATUS "Non-Windows: portable core tests only; no Windows application is built")
  return()
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "LiteReplay requires a 64-bit build")
endif()
# vcpkg ships FindFFMPEG; manual SDKs can supply include/, lib/ and bin/.
set(FFMPEG_ROOT "" CACHE PATH "Optional shared FFmpeg SDK root")
if(FFMPEG_ROOT)
  find_path(FFMPEG_INCLUDE_DIRS libavcodec/avcodec.h PATHS "${FFMPEG_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  foreach(component avcodec avformat avutil swresample)
    find_library(FFMPEG_${component} NAMES ${component} PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    list(APPEND FFMPEG_LIBRARIES "${FFMPEG_${component}}")
  endforeach()
else()
  find_package(FFMPEG REQUIRED)
endif()
add_library(replay_engine STATIC src/video.cpp src/audio.cpp src/engine.cpp)
target_include_directories(replay_engine PUBLIC src ${FFMPEG_INCLUDE_DIRS})
target_compile_definitions(replay_engine PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_directories(replay_engine PUBLIC ${FFMPEG_LIBRARY_DIRS})
target_link_libraries(replay_engine PUBLIC ${FFMPEG_LIBRARIES} d3d11 dxgi dxguid ole32 avrt uuid ksuser)
add_executable(LiteReplay WIN32 src/app.cpp)
target_link_libraries(LiteReplay PRIVATE replay_engine windowscodecs mfuuid comctl32 shell32 advapi32 dwmapi psapi pdh winmm comdlg32 wbemuuid oleaut32)
add_executable(LiteReplayProbe tests/integration_win.cpp)
target_link_libraries(LiteReplayProbe PRIVATE replay_engine)
if(MSVC)
  foreach(target replay_engine LiteReplay LiteReplayProbe)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive- /EHsc)
  endforeach()
endif()
# vcpkg's app-local deployment copies runtime DLLs beside both executables.
if(FFMPEG_ROOT)
  file(GLOB FFMPEG_DLLS "${FFMPEG_ROOT}/bin/*.dll")
  if(NOT FFMPEG_DLLS)
    message(FATAL_ERROR "Manual FFmpeg SDK must contain shared DLLs in bin/")
  endif()
  foreach(target LiteReplay LiteReplayProbe)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${FFMPEG_DLLS} "$<TARGET_FILE_DIR:${target}>" COMMAND_EXPAND_LISTS)
  endforeach()
endif()

add_custom_command(TARGET LiteReplay POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/bin/windows-x64/save.wav" "$<TARGET_FILE_DIR:LiteReplay>/save.wav")

add_custom_command(TARGET LiteReplay POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/languages" "$<TARGET_FILE_DIR:LiteReplay>/languages")

add_custom_command(TARGET LiteReplay POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/bin/windows-x64/editor-tools" "$<TARGET_FILE_DIR:LiteReplay>/editor-tools")
add_custom_command(TARGET LiteReplay POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/bin/windows-x64/performance-tools" "$<TARGET_FILE_DIR:LiteReplay>/performance-tools")
